autonumber

skinparam sequenceReferenceBorderColor #4682B4

actor "Utilisateur" as User #FFD700
participant "Interface Web" as UI #87CEEB
participant "API / Passerelle" as API #ADD8E6
participant "File d'Attente\nde Tâches" as Queue #DDA0DD
participant "Worker d'Ingestion" as Worker #98FB98
participant "Système de\nCompréhension Docs" as Parser #FFB6C1
database "Base de Données\nRelationnelle" as RelDB #F08080
database "Index Vectoriel" as VecDB #87CEFA
database "Base Graphe" as GraphDB #D3D3A0

title Diagramme de séquence - Horloge A : Ingestion d'un document (Architecture Générique)

== Étape 1 : Téléversement synchrone (Réponse rapide) ==
User -> UI : Envoie le document (DAO, référence, etc.)
activate UI #87CEEB
UI -> API : POST /documents (Fichier + Métadonnées)
activate API #ADD8E6
API -> API : Valide les droits d'accès\net le tenant
note right of API : Si l'utilisateur n'a pas les droits,\nl'ingestion est refusée ici.

API -> RelDB : Créer Document (status: "EN_ATTENTE")
activate RelDB #F08080
RelDB --> API : ID_Document
deactivate RelDB

API -> Queue : Déposer tâche "INGERER_DOCUMENT"\n(ID du document)
activate Queue #DDA0DD
Queue --> API : Accusé de réception
deactivate Queue

API --> UI : Accusé de réception (ID du document)
deactivate API
UI --> User : "Document reçu,\ntraitement en cours..."
deactivate UI

note right of Queue #FFFFE0
  <b>L'utilisateur peut</b>
  continuer à travailler
  pendant le traitement
end note

== Étape 2 : Traitement asynchrone (Horloge A) ==

Worker -> Queue : Consomme la tâche "INGERER_DOCUMENT"
activate Worker #98FB98
activate Queue #DDA0DD
Queue --> Worker : Tâche (ID du document)
deactivate Queue

Worker -> RelDB : Mettre à jour status: "EN_COURS"
activate RelDB #F08080
RelDB --> Worker : OK
deactivate RelDB

alt Document est un PDF numérique
    Worker -> Parser : Demander extraction structurée
    activate Parser #FFB6C1
    Parser -> Parser : Extraction des titres,\ntableaux et mise en forme
    Parser --> Worker : Contenu structuré (Markdown, tables)
    deactivate Parser
else Document est un Scan/Image
    Worker -> Parser : Demander OCR + extraction
    activate Parser #FFB6C1
    Parser -> Parser : Reconnaissance optique\net structure du document
    Parser --> Worker : Texte OCR + métadonnées
    deactivate Parser
else Document est DOCX/XLSX
    Worker -> Parser : Demander extraction directe
    activate Parser #FFB6C1
    Parser -> Parser : Lecture du format bureautique
    Parser --> Worker : Contenu et structure
    deactivate Parser
else Format non supporté
    Worker -> Worker : Déclencher une erreur
    Worker -> RelDB : Mettre à jour status: "ECHEC"
    activate RelDB #F08080
    RelDB --> Worker : OK
    deactivate RelDB
    Worker -> Queue : Notifier l'échec
    activate Queue #DDA0DD
    Queue --> Worker : OK
    deactivate Queue
    note right of Worker #FFE4E1
      <b>⚠️ Échec d'ingestion</b>
      L'utilisateur sera notifié
      et pourra retéléverser
    end note
end

== Étape 3 : Découpage et indexation ==

Worker -> Worker : Découpage en "chunks" sémantiques\n(avec hiérarchie parent/enfant)
note right of Worker #FFFFE0
  <b>Règles de découpage :</b>
  • Parent : article ou section (max 8000 car)
  • Enfant : paragraphe (1100 car, overlap 120)
  • Titres → hiérarchie préservée
end note

Worker -> RelDB : Sauvegarder les chunks texte\n(contenu, page, titre, parent_id)
activate RelDB #F08080
RelDB --> Worker : OK
deactivate RelDB

Worker -> VecDB : Indexer les vecteurs des chunks
activate VecDB #87CEFA
note right of VecDB #FFFFE0
  <b>Indexation vectorielle :</b>
  • Génération des embeddings
  • Indexation dense pour recherche
    sémantique (cosine similarity)
end note
VecDB --> Worker : OK
deactivate VecDB

Worker -> RelDB : Mettre à jour status: "PRET"
activate RelDB #F08080
RelDB --> Worker : OK
deactivate RelDB

note over Worker, RelDB #98FB98
  <b>✅ Document prêt pour le RAG</b>
  L'utilisateur peut interroger
  et générer des sections
end note

== Étape 4 : Extraction du graphe (Phase 1.5) ==

note right of Worker #FFFFE0
  <b>Phase non bloquante</b>
  Le RAG fonctionne déjà sans elle
end note

Worker -> RelDB : Charger les chunks parents\ndu document (max 12)
activate RelDB #F08080
RelDB --> Worker : Liste des chunks parents
deactivate RelDB

loop Pour chaque chunk parent
    Worker -> Worker : Construire un prompt d'extraction\n(Appel LLM pour entités/relations)
    note right of Worker #FFFFE0
      <b>Extraction LLM :</b>
      • Entités : ENTREPRISE, PROJET,
        CLIENT, CERTIFICATION, PERSONNE
      • Relations : A_REALISE, POUR,
        DETIENT, SATISFAITE_PAR
      • Sortie JSON contraint
    end note
    
    Worker -> Worker : Résolution d'entités\n(exact, fuzzy, blocking)
    note right of Worker #FFFFE0
      <b>Éviter les doublons :</b>
      1. Match exact (SIRET, ID interne)
      2. Nom normalisé
      3. Fuzzy (bloc seulement)
      4. Création si non trouvé
    end note
    
    Worker -> GraphDB : Enregistrer les entités\net relations (MERGE)
    activate GraphDB #D3D3A0
    GraphDB --> Worker : OK
    deactivate GraphDB
end

Worker -> RelDB : Marquer document comme 'graphe_extrait' (true)
activate RelDB #F08080
RelDB --> Worker : OK
deactivate RelDB

deactivate Worker

note over User, GraphDB #F0F8FF
  <b>✅ Horloge A terminée</b>
  • RAG (recherche textuelle) : disponible immédiatement
  • Graphe (relations entre entités) : disponible après extraction
end note

== Étape 5 : Notification finale (optionnelle) ==

User -> UI : Vérifier l'état du document
activate UI #87CEEB
UI -> API : GET /documents/{id}/status
activate API #ADD8E6
API -> RelDB : Vérifier le statut du document
activate RelDB #F08080
RelDB --> API : Statut: "PRET"
deactivate RelDB
API --> UI : Document prêt
deactivate API
UI --> User : "Document disponible"
deactivate UI

title Diagramme de séquence - Horloge B : Génération d'une section

User -> UI : Demande la génération\nd'une section (ex: "Références")
activate UI

activate API
API -> Orchestrator : Délègue la tâche
activate Orchestrator

Orchestrator -> DB : Récupère le contexte du dossier\n(tenant, plan, exigences)
activate DB
DB --> Orchestrator : Contexte du dossier
deactivate DB

== ÉTAPE 2 : RECHERCHE DES SOURCES ==
Orchestrator -> Orchestrator : Construit la requête de recherche\n(filtres : tenant, rôles documents)

activate VecDB

    activate GraphDB
    GraphDB --> Orchestrator : Entités et relations
    deactivate GraphDB

activate LLM
LLM --> Orchestrator : Contenu + citations
deactivate LLM

Orchestrator -> Orchestrator : Valide le résultat\n(citations existantes, structure)

== ÉTAPE 4 : SAUVEGARDE ET REVUE HUMAINE ==
Orchestrator -> DB : Sauvegarde la version générée\n(generations)
activate DB
DB --> Orchestrator : OK
deactivate DB

deactivate Orchestrator

deactivate API

deactivate UI

  HITL (Human In The Loop)
  Le rédacteur valide avant intégration
end note

User -> UI : Approuve, modifie ou demande\nune régénération
activate UI
UI -> API : POST /reviews (action + commentaire)
activate API
API -> Orchestrator : Traite la décision humaine
activate Orchestrator

alt Action = "Approuver"
    Orchestrator -> DB : Statut de la section → "APPROUVEE"
    activate DB
    DB --> Orchestrator : OK
    deactivate DB
    Orchestrator --> API : Section validée
    deactivate Orchestrator
    API --> UI : Confirmation
    deactivate API
    UI --> User : "Section approuvée"
    deactivate UI

else Action = "Régénérer"
    Orchestrator -> Orchestrator : Relance la génération\n(ÉTAPE 2) avec nouvelle consigne
    note right of Orchestrator #FFFFE0
      Boucle de régénération
      jusqu'à approbation
    end note

else Action = "Modifier"
    Orchestrator -> DB : Sauvegarde le texte modifié
    activate DB
    DB --> Orchestrator : OK
    deactivate DB
    Orchestrator -> DB : Statut → "APPROUVEE"
    activate DB
    DB --> Orchestrator : OK
    deactivate DB
    Orchestrator --> API : Section modifiée
    deactivate Orchestrator
    API --> UI : Confirmation
    deactivate API
    UI --> User : "Section modifiée et approuvée"
    deactivate UI
end

note over User, GraphDB #F0F8FF
  ✅ Section prête pour l'export
  L'ensemble des sections approuvées
  peut être assemblé en document final
end note
