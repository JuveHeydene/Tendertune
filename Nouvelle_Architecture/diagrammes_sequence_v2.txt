##### DIAGRAMME 3 — HORLOGE B V2 : BOUCLE CRAG DÉTAILLÉE (pour UN expert)


@startuml
skinparam backgroundColor white
skinparam defaultFontName Arial
skinparam sequenceMessageAlign center
autonumber

skinparam ArrowColor #333333
skinparam sequenceLifeLineBorderColor #999999

participant "Superviseur /\nOrchestrateur" as Orchestrator #FFA07A

participant "Reranker" as Reranker #DDA0DD
participant "Évaluateur\n(grading)" as Grader #F5DEB3
database "retrieval_traces" as Traces #C0C0C0

title Diagramme de séquence V2 — Boucle CRAG bornée (Adaptive + grading + abstention)

Orchestrator -> Expert : sous_question.texte, filtres\n(section_type + doc_role, cf. §3.1)
activate Expert #B0E0E6

== Adaptive : décision AVANT toute recherche (§5.1) ==
Expert -> Expert : complexite ?
alt complexite = "simple"
    Expert -> VecDB : retrieve_dense(query, filtres, top_k=8)
    activate VecDB #87CEFA
    VecDB --> Expert : Top-8
    deactivate VecDB
    Expert -> Reranker : rerank
    activate Reranker #DDA0DD
    Reranker --> Expert : Top-8 ordonné
    deactivate Reranker
    note right of Expert #E0FFFF
      PAS de grading ici (§5.1) —
      1 seule passe, on fait confiance
      au Top-8 (comme la V1, + hybride)
    end note
    Expert --> Orchestrator : contexte (source="retrieval")
else complexite = "complexe"
    == Boucle CRAG, tentative 1..3 (max_iter validé) ==
    loop tentative = 1 à 3
        par Recherche hybride en parallèle
            Expert -> VecDB : retrieve_dense(query, filtres, top_k=50)
            activate VecDB #87CEFA
            VecDB --> Expert : 50 candidats (dense)
            deactivate VecDB
        and
            Expert -> FTS : retrieve_sparse(query, filtres, top_k=50)
            activate FTS #F0A0A0
            note right of FTS #FFFFE0
              Index inversé (tsvector/GIN,
              config "french") — PAS un scan
              complet, lookup par terme
            end note
            FTS --> Expert : 50 candidats (sparse)
            deactivate FTS
        end
        Expert -> Expert : fusion RRF (même granularité :\nenfants des deux côtés, §4.4)
        Expert -> Reranker : rerank → Top-8
        activate Reranker #DDA0DD
        Reranker --> Expert : Top-8
        deactivate Reranker

        Expert -> Grader : grade_context(query, Top-8)
        activate Grader #F5DEB3
        Grader -> Grader : seuil de score du reranker\n(> 0.7 / 0.3-0.7 / < 0.3)
        alt zone grise (0.3-0.7)
            Grader -> Grader : LLM-juge (1 appel, seulement ici)
        end
        Grader --> Expert : Correct | Ambigu | Incorrect
        deactivate Grader

        Expert -> Traces : INSERT retrieval_traces\n(expert, tentative, query, grade, chunks)
        activate Traces #C0C0C0
        Traces --> Expert : OK
        deactivate Traces

        alt grade = "Correct"
            Expert -> Expert : sort de la boucle (succès)
        else grade = "Ambigu"
            Expert -> Expert : ÉLARGIR (pas reformuler)\ndrop section_type, drop doc_role,\nou top_k 50→100
        else grade = "Incorrect"
            Expert -> Expert : REFORMULER (rewrite variante\nou multi-query, RAG-08 §4)
        end
    end

    alt Une tentative a grade="Correct"
        Expert --> Orchestrator : contexte gradé (source="retrieval")
    else 3 tentatives épuisées, jamais Correct
        Expert -> Expert : ABSTENTION explicite
        note right of Expert #FFE4E1
          { found: false, reason: "...",
            tentatives: 3 }
          → PAS de génération sur du vide
          (SectionGeneree.abstentions[])
        end note
        Expert --> Orchestrator : abstention (source="abstention")
    end
end
deactivate Expert

legend right
  | <color:#FFA07A>Superviseur</color> | A délégué cette sous-question |
  | <color:#B0E0E6>Expert</color> | Scope = section_type + doc_role (§1.1) |
  | <color:#87CEFA>Index Vectoriel</color> | Qdrant — HNSW, approché, pas exhaustif |
  | <color:#F0A0A0>Postgres FTS</color> | tsvector/GIN — index inversé |
  | <color:#DDA0DD>Reranker</color> | Cross-encoder, Top-50 → Top-8 |
  | <color:#F5DEB3>Évaluateur</color> | Seuil + LLM-juge en zone grise |
  | <color:#C0C0C0>retrieval_traces</color> | 1 ligne par tentative — observabilité + éval mémoire |

    ExpertGraph -> Fuzzy : fuzzy_resolve_entity(name, type_hint, tenant_id)
    activate Fuzzy #FFDAB9
    Fuzzy -> Fuzzy : exact → Levenshtein ≥85\n→ Jaccard ≥80 → embedding ≥0.82
    Fuzzy --> ExpertGraph : clé canonique OU "non résolu"
    deactivate Fuzzy

alt Aucune entité résolue
    ExpertGraph -> ExpertGraph : court-circuit immédiat
    note right of ExpertGraph #FFE4E1
      Pas d'appel Cypher inutile —

    end note
    ExpertGraph --> Orchestrator : abstention

    ExpertGraph -> ExpertGraph : sélectionne un TEMPLATE Cypher fixe\n(1-hop générique | 1-2 sauts typés | 2 ancres)
    note right of ExpertGraph #FFFFE0
      Jamais de Cypher généré librement
      par le LLM — le code construit la
      requête à partir de templates

      "un agent, une responsabilité")
    end note
    ExpertGraph -> GraphDB : exécute le template paramétré
    activate GraphDB #D3D3A0
    GraphDB --> ExpertGraph : relations trouvées
    deactivate GraphDB

    alt Relations trouvées
        ExpertGraph -> RelDB : SELECT text, page FROM chunks\nWHERE id = source_chunk_id
        activate RelDB #F08080

        deactivate RelDB
        ExpertGraph --> Orchestrator : { relations[], chunks_source[] }\n(source="retrieval")
    else Rien trouvé malgré entités résolues
        ExpertGraph -> ExpertGraph : élargir le hop (1→2) avant abandon
        ExpertGraph --> Orchestrator : abstention si toujours rien
    end
end
deactivate ExpertGraph
